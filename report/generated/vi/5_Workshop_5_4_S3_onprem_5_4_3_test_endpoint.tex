\vspace{0.5em}\noindent\textbf{Lấy regional DNS name (tên DNS khu vực) của S3 interface
endpoint}\par

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\item
  Trong Amazon VPC menu, chọn Endpoints.
\item
  Click tên của endpoint chúng ta mới tạo ở mục 4.2:
  s3-interface-endpoint. Click details và lưu lại regional DNS name của
  endpoint (cái đầu tiên) vào text-editor của bạn để dùng ở các bước
  sau.
\end{enumerate}

\vspace{0.5em}\noindent\textbf{Kết nối đến EC2 instance ở trong ``VPC On-prem'' (giả lập môi
trường truyền
thống)}\par

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\item
  Đi đến \textbf{Session manager} bằng cách gõ ``session manager'' vào ô
  tìm kiếm
\item
  Click \textbf{Start Session}, chọn EC2 instance có tên
  \textbf{Test-Interface-Endpoint}. EC2 instance này đang chạy trên
  ``VPC On-prem'' và sẽ được sử dụng để kiểm tra kết nối đến Amazon S3
  thông qua Interface endpoint. Session Manager sẽ mở 1 browser tab mới
  với shell prompt: \textbf{sh-4.2 \$}
\end{enumerate}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\setcounter{enumi}{4}
\tightlist
\item
  Copy file vào S3 bucket mình tạo ở section 4.2
\end{enumerate}

\begin{verbatim}
aws s3 cp --endpoint-url https://bucket.<Regional-DNS-Name> testfile2.xyz s3://<your-bucket-name>
\end{verbatim}

\begin{itemize}
\tightlist
\item
  Câu lệnh này yêu cầu thông số --endpoint-url, bởi vì bạn cần sử dụng
  DNS name chỉ định cho endpoint để truy cập vào S3 thông qua Interface
  endpoint.
\item
  Không lấy ' * ' khi copy/paste tên DNS khu vực.
\item
  Cung cấp tên S3 bucket của bạn
\end{itemize}
